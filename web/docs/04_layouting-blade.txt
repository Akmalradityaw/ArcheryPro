                @include('layouts.partials.navbar')
                @include('layouts.partials.alert')
                <main class="p-4 lg:p-6 flex-1">
                    @yield('content')
                </main>
                @include('layouts.partials.footer')
            </div>

            <div class="drawer-side">
                @include('layouts.partials.sidebar')
            </div>
        </div>

        <script src="{{ asset('build/app.js') }}"></script>

    </body>
    </html>

--------------------------------------------------------------------------------
E. SIDEBAR DINAMIS PER ROLE
--------------------------------------------------------------------------------
layouts/partials/sidebar.blade.php memuat menu berbeda berdasarkan role:

    @auth
        @if(auth()->user()->hasRole('admin'))
            @include('layouts.partials.menu-admin')
        @elseif(auth()->user()->hasRole('pelatih'))
            @include('layouts.partials.menu-pelatih')
        @elseif(auth()->user()->hasRole('scoring'))
            @include('layouts.partials.menu-scoring')
        @elseif(auth()->user()->hasRole('atlet'))
            @include('layouts.partials.menu-atlet')
        @endif
    @endauth

--------------------------------------------------------------------------------
F. COMPONENT BLADE YANG WAJIB DIBUAT
--------------------------------------------------------------------------------
1. x-card-stat           -> card statistik dashboard
2. x-btn-primary         -> tombol hijau solid konsisten
3. x-table-empty         -> placeholder tabel kosong
4. x-modal-confirm       -> modal konfirmasi hapus
5. x-input-error         -> pesan error validasi
6. x-alert               -> alert flash message (success/error/warning)
7. x-badge-status        -> badge status event/atlet
8. x-checkbox-atlet      -> checkbox multi-atlet khusus scoring
